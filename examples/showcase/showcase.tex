\newcommand{\ShowSlide}[7]{%

  % Draw in Beamer's page-sized background, without remembered coordinates.

    \begin{tikzpicture}[x=1cm,y=1cm]
      \path[use as bounding box] (0,0) rectangle (16,9);
      \begin{scope}
        \clip (0,0) rectangle (16,9);
        \fill[white,opacity=.035] (7.5,0) rectangle (16,9);
        % Fine full-height lines make motion and page boundaries easy to see.
        \foreach \x in {8,9,...,16} \draw[white,opacity=.07] (\x,0)--(\x,9);
        \pgfmathtruncatemacro{\kind}{mod(#5,5)}
        \begin{scope}[shift={(11.8,4.65)},rotate={mod(#5*17,70)-35}]
        \ifcase\kind
          % Orbital system: concentric ellipses and an offset planet.
          \foreach \r in {1.1,1.7,2.3,2.9}
            \draw[cream,opacity=.5,line width=1pt] (0,0) ellipse ({\r} and {0.6*\r});
          \fill[coral] (0,0) circle (1.03);
          \fill[gold] (2.1,.95) circle (.42);
          \fill[mint] (-2,-.8) circle (.24);
          \draw[cream,line width=2pt] (-.5,-2.65)--(1.4,-2.65);
        \or
          % Faceted cube: three readable, contrasting faces.
          \fill[mint] (0,2.4)--(2.5,1)--(0,-.4)--(-2.5,1)--cycle;
          \fill[coral] (-2.5,1)--(0,-.4)--(0,-3)--(-2.5,-1.6)--cycle;
          \fill[gold] (0,-.4)--(2.5,1)--(2.5,-1.6)--(0,-3)--cycle;
          \draw[cream,line width=1pt] (0,2.4)--(2.5,1)--(2.5,-1.6)--(0,-3)--(-2.5,-1.6)--(-2.5,1)--cycle;
          \foreach \x in {-3,-2,...,3} \fill[cream,opacity=.6] (\x,3.1) circle (.035);
        \or
          % Layered landscape: sun, distant ridges and foreground.
          \fill[gold] (1.3,1.4) circle (1.1);
          \fill[mint!60!#6] (-3,-1.7)--(-1.4,1.5)--(.2,-.6)--(1.1,.6)--(3,-1.7)--cycle;
          \fill[mint] (-3,-2.4)--(-1,-.3)--(.4,-1.5)--(2,.3)--(3,-2.4)--cycle;
          \fill[coral] (-3,-2.9)--(-1.5,-1.7)--(.4,-2.3)--(2,-1.6)--(3,-2.9)--cycle;
          \foreach \y in {-3.2,-3.45,-3.7} \draw[cream,opacity=.4] (-3,\y)--(3,\y);
        \or
          % Parametric ribbon: phase varies on every appearance.
          \foreach \i in {0,...,15} {
            \pgfmathtruncatemacro{\mix}{\i*5}
            \draw[color=mint!\mix!coral,line width=2pt]
              plot[domain=-3:3,samples=60] (\x,{1.5*sin(55*\x+#5*11)+\i*.16-1.2});
          }
          \fill[gold] (2.2,2.4) circle (.35);
        \or
          % Radial burst: large asymmetric silhouette with fine spokes.
          \foreach \a in {0,15,...,345} {
            \draw[cream,opacity=.5,line width=1pt] (\a:1.2)--(\a:{2.5+.3*sin(\a*3+#5)});
            \fill[mint] (\a:2.75) circle (.06);
          }
          \fill[coral] (0,0) circle (1.15);
          \fill[gold] (.35,.3) circle (.42);
          \draw[cream,line width=2pt] (-2.9,-2.7)--(-1.1,-2.7);
        \fi
        \end{scope}
        \node[anchor=north west,text=cream,font=\small\bfseries] at (.8,8.35) {PDFCUBE / TRANSITION SHOWCASE};
        \node[anchor=north west,text=cream,text width=6.2cm,align=left,
              font=\fontsize{24}{28}\selectfont\bfseries] at (.8,7.25) {#2};
        \node[anchor=north west,text=cream,text width=6cm,align=left,
              font=\fontsize{14}{19}\selectfont] at (.8,4.05) {#3};
        \node[anchor=south west,text=mint,text width=6.2cm,align=left,font=\footnotesize] at (.8,2.15) {#7};
        \draw[cream,opacity=.35] (.8,1.6)--(15.2,1.6);
        \node[anchor=north west,text=cream,font=\small,text width=13cm,align=left] at (.8,1.3) {\textbf{Next:} #4};
        \node[anchor=south west,text=cream!65!#6,font=\scriptsize] at (.8,.35) {SPACE / PAGE DOWN TO ADVANCE};
        \node[anchor=south east,text=cream,font=\small] at (15.2,.35) {\insertframenumber};
      \end{scope}
    \end{tikzpicture}%
  }%
  \begin{frame}[plain]
    \hypersetup{pdfpagetransition={#1}}%
    \transduration{\ShowcaseDwell}%
    \null
  \end{frame}%
